2010   LOCAL i
2020   FOR i = 1 TO n: LET a(i) = INT(RND * 1000): NEXT i

4000 DEFPROC shellSort(n)
4010   LOCAL g, i, j, t
4020   LET g = INT(n / 2): REM Set the initial gap
4030   REPEAT : WHILE g > 0
4040     FOR i = g + 1 TO n
4050       LET t = a(i)
4060       LET j = i
4070       REPEAT
4080         WHILE j > g
4090         WHILE a(j - g) > t
4100         LET a(j) = a(j - g)
4110         LET j = j - g
4120       REPEAT UNTIL 0
4130       LET a(j) = t
4140     NEXT i
4150     LET g = INT(g / 2)
4160   REPEAT UNTIL 0
4170 ENDPROC
