10 REM ==============BERNOULLI.BAS====================
20 REM  BERNOULLI NUMBERS (B0..BN) - NextBASIC
30 REM  Recreates Ada Lovelace's algorithm
40 REM  Bm ends up in A(1) at the end of each iteration m
50 REM ================================================

60 LET N = 14
70 DIM A(N+1)

80 PRINT "Bernoulli numbers B0..B";N
90 PRINT "--------------------------------"

100 FOR m = 0 TO N

110   REM A(m) <- 1/(m+1)  (we use A(m+1) because indices are 1-based)
120   LET A(m+1) = 1/(m+1)

130   REM Backward sweep: A(j-1) <- j*(A(j-1)-A(j))
140   FOR j = m+1 TO 2 STEP -1
150     LET A(j-1) = (j-1) * (A(j-1) - A(j))
160   NEXT j

170   REM At this point, A(1) = B_m
180   PRINT "B";m;" = ";A(1)

190 NEXT m

200 PRINT "--------------------------------"
210 PRINT "Note: for odd m > 1, Bm should be 0 (almost)."
220 PRINT "If you raise N a lot, floating-point rounding shows."
